> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship the v0.6.0 GitHub auto-backup: `urika github init/push/pull/status`, opt-in `auto_push` hook firing after `run`/`finalize`/`build-tool`, pre-commit secret scanner, researcher-safe `.gitignore`, dashboard Git tab. Zero auth surface — Urika never sees a GitHub credential.

**Architecture:** Single `core/git.py` module wraps `git` (and optionally `gh`) via `subprocess`, encapsulating init/commit/push/pull/status, secret scanning, and `.gitignore` management. CLI subcommands in `cli/github.py` are thin Click handlers that route to the core module. Dashboard adds 4 read-only/write API routes and one Jinja template. Auto-push is a `_maybe_auto_push(project_dir)` helper called at the tail of three existing CLI commands; the orchestrator stays untouched.

**Tech Stack:** Python 3.11+, `subprocess` (standard library), Click, FastAPI, Jinja2 + HTMX + Alpine. Optional `gh` CLI for `--create`. pytest with a tmp_path bare-remote fixture for unit tests; one opt-in integration test in `tests/test_integration_github_push.py`.

```python
# src/urika/core/git.py
"""Pure ``git`` shell-out wrapper for the v0.6.0 GitHub auto-backup feature.

Design:

- All functions are stateless wrappers around the system ``git`` binary
  invoked via ``subprocess.run``. No in-memory state, no caching.
- Errors come back as values (boolean ``ok`` + ``error`` message in
  result dataclasses) rather than exceptions. Callers print remediation
  hints cleanly.
- The user's existing ``git`` auth (HTTPS+PAT, SSH key, gh-stored
  credential — anything that makes ``git push`` work in their shell)
  does all the authentication. Urika never sees a credential.

See ``dev/plans/2026-05-31-v0.6.0-github-design.md`` for the locked
design decisions.
"""

from __future__ import annotations

import subprocess
from pathlib import Path


def _git(cwd: Path, *args: str, check: bool = True) -> subprocess.CompletedProcess[str]:
    """Run ``git`` with ``cwd`` set to *cwd*. Returns the completed process.

    ``check=True`` (default) raises ``CalledProcessError`` on non-zero
    exit. Callers that want to inspect failures themselves pass
    ``check=False``.
    """
    return subprocess.run(
        ["git", *args],
        cwd=str(cwd),
        capture_output=True,
        text=True,
        check=check,
    )


def is_git_repo(project_dir: Path) -> bool:
    """True if *project_dir* contains a ``.git`` directory at its root."""
    return (project_dir / ".git").is_dir()


def init_repo(project_dir: Path) -> None:
    """Initialise *project_dir* as a git repo. Idempotent."""
    if is_git_repo(project_dir):
        return
    _git(project_dir, "init", "-q")


def add_remote(project_dir: Path, url: str, name: str = "origin") -> None:
    """Set the remote *name* to *url*. Creates if absent, updates if present."""
    result = _git(project_dir, "remote", "get-url", name, check=False)
    if result.returncode == 0:
        _git(project_dir, "remote", "set-url", name, url)
    else:
        _git(project_dir, "remote", "add", name, url)
```

```python
_GITIGNORE_TEMPLATE = """\
# Urika internal state — never commit (secrets, sessions, meta)
.urika/
*.lock

# Python
__pycache__/
*.pyc
.venv/
venv/
.tox/

# Credentials / private keys (never commit, even if you remove other
# patterns from this list)
.env
*.pem
*.key
*_rsa
*_ed25519

# Research data — researcher-safe default exclusion. Remove the lines
# below (or override via [github] gitignore_extra in urika.toml) if
# your project's raw data is safe to commit.
data/raw/
*.parquet
*.h5
*.pkl
*.joblib
"""

_BLOCK_START = "# >>> urika: managed block — do not edit between markers >>>"
_BLOCK_END = "# <<< urika: managed block <<<"


def write_gitignore(project_dir: Path, extra_patterns: list[str] | None = None) -> None:
    """Write or update the project's ``.gitignore`` with researcher-safe defaults.

    Behaviour:
    - Fresh project: writes a new ``.gitignore`` containing the managed
      block (between BLOCK_START / BLOCK_END markers) and no other content.
    - Existing ``.gitignore`` without an Urika block: appends the
      managed block at the end, preserving user content.
    - Existing ``.gitignore`` WITH an Urika block: replaces the block
      in place, preserving content above and below it.

    ``extra_patterns`` are user-provided lines appended inside the
    managed block. Comes from ``[github] gitignore_extra`` in urika.toml.
    """
    extra = list(extra_patterns or [])
    block_lines = [_BLOCK_START, _GITIGNORE_TEMPLATE.rstrip("\n")]
    if extra:
        block_lines.append("\n# Extra patterns from urika.toml")
        block_lines.extend(extra)
    block_lines.append(_BLOCK_END)
    block = "\n".join(block_lines) + "\n"

    path = project_dir / ".gitignore"
    if not path.exists():
        path.write_text(block, encoding="utf-8")
        return

    existing = path.read_text(encoding="utf-8")
    if _BLOCK_START in existing and _BLOCK_END in existing:
        # Replace the existing block in place.
        before, _, rest = existing.partition(_BLOCK_START)
        _old_block, _, after = rest.partition(_BLOCK_END)
        new_text = before.rstrip() + "\n\n" + block + after.lstrip()
        # Collapse runs of >=3 newlines that the slicing may have produced.
        while "\n\n\n" in new_text:
            new_text = new_text.replace("\n\n\n", "\n\n")
        path.write_text(new_text, encoding="utf-8")
    else:
        # Append the block at the end, preserving everything else.
        sep = "" if existing.endswith("\n") else "\n"
        path.write_text(existing + sep + "\n" + block, encoding="utf-8")
```

```python
from dataclasses import dataclass


@dataclass
class CommitResult:
    ok: bool
    sha: str | None = None
    error: str | None = None


@dataclass
class PushResult:
    ok: bool
    pushed_sha: str | None = None
    error: str | None = None


@dataclass
class PullResult:
    ok: bool
    error: str | None = None


@dataclass
class GitStatus:
    branch: str
    dirty_count: int
    remote_url: str | None
    last_commit_sha: str | None
    last_commit_subject: str | None


@dataclass
class GitCommit:
    sha: str
    subject: str
    event: str | None  # populated from ``urika-event:`` trailer when present
    ts: str            # ISO 8601


def commit_all(project_dir: Path, message: str) -> CommitResult:
    """Stage everything and create a commit. Returns ok=False if there's
    nothing to commit (clean working tree)."""
    _git(project_dir, "add", "-A")
    # Detect "nothing to commit" without raising.
    diff = _git(project_dir, "diff", "--cached", "--name-only", check=False)
    if diff.returncode == 0 and not diff.stdout.strip():
        return CommitResult(ok=False, error="nothing to commit (working tree clean)")
    result = _git(project_dir, "commit", "-m", message, check=False)
    if result.returncode != 0:
        return CommitResult(ok=False, error=result.stderr.strip() or result.stdout.strip())
    sha = _git(project_dir, "rev-parse", "HEAD").stdout.strip()
    return CommitResult(ok=True, sha=sha[:12])


def push(project_dir: Path, remote: str = "origin") -> PushResult:
    """Push current branch to *remote*. ok=False with verbatim git stderr on failure."""
    result = _git(project_dir, "push", remote, "HEAD", check=False)
    if result.returncode != 0:
        return PushResult(ok=False, error=(result.stderr or result.stdout).strip())
    sha = _git(project_dir, "rev-parse", "HEAD").stdout.strip()
    return PushResult(ok=True, pushed_sha=sha[:12])


def pull(project_dir: Path, remote: str = "origin") -> PullResult:
    """Fetch + ff-only merge from *remote*. Refuses on divergence."""
    fetch = _git(project_dir, "fetch", remote, check=False)
    if fetch.returncode != 0:
        return PullResult(ok=False, error=(fetch.stderr or fetch.stdout).strip())
    merge = _git(project_dir, "merge", "--ff-only", "FETCH_HEAD", check=False)
    if merge.returncode != 0:
        return PullResult(ok=False, error=(merge.stderr or merge.stdout).strip())
    return PullResult(ok=True)


def status(project_dir: Path) -> GitStatus:
    """Snapshot of current branch, dirty file count, remote URL, last commit."""
    branch = _git(project_dir, "rev-parse", "--abbrev-ref", "HEAD", check=False)
    branch_name = branch.stdout.strip() if branch.returncode == 0 else ""
    porcelain = _git(project_dir, "status", "--porcelain", check=False)
    dirty = (
        len([line for line in porcelain.stdout.splitlines() if line.strip()])
        if porcelain.returncode == 0
        else 0
    )
    remote = _git(project_dir, "remote", "get-url", "origin", check=False)
    remote_url = remote.stdout.strip() if remote.returncode == 0 else None
    last = _git(
        project_dir, "log", "-1", "--format=%H%n%s", check=False
    )
    last_sha = None
    last_subject = None
    if last.returncode == 0 and last.stdout.strip():
        parts = last.stdout.strip().split("\n", 1)
        last_sha = parts[0][:12]
        last_subject = parts[1] if len(parts) > 1 else ""
    return GitStatus(
        branch=branch_name,
        dirty_count=dirty,
        remote_url=remote_url,
        last_commit_sha=last_sha,
        last_commit_subject=last_subject,
    )


def recent_commits(project_dir: Path, limit: int = 10) -> list[GitCommit]:
    """Last *limit* commits, newest first. Parses ``urika-event:`` from the body."""
    fmt = "%H%x00%s%x00%aI%x00%b%x1e"  # NUL-separated fields, RS-separated records
    out = _git(
        project_dir,
        "log",
        f"-{limit}",
        f"--format={fmt}",
        check=False,
    )
    if out.returncode != 0 or not out.stdout.strip():
        return []
    commits: list[GitCommit] = []
    for rec in out.stdout.split("\x1e"):
        rec = rec.strip("\n")
        if not rec:
            continue
        fields = rec.split("\x00")
        if len(fields) < 4:
            continue
        sha, subject, ts, body = fields[0], fields[1], fields[2], fields[3]
        event = None
        for line in body.splitlines():
            if line.startswith("urika-event:"):
                event = line.split(":", 1)[1].strip()
                break
        commits.append(GitCommit(sha=sha[:12], subject=subject, event=event, ts=ts))
    return commits
```

```python
import re


@dataclass
class SecretMatch:
    file: str
    line: int
    pattern_name: str
    preview: str  # masked: first 8 + last 4 chars of the matched token


# Patterns ordered most-specific first so e.g. ``sk-ant-`` doesn't get
# overshadowed by the generic ``sk-`` openai pattern.
_SECRET_PATTERNS: list[tuple[str, re.Pattern[str]]] = [
    ("anthropic_api_key", re.compile(r"sk-ant-[A-Za-z0-9_\-]{32,}")),
    ("github_pat", re.compile(r"ghp_[A-Za-z0-9]{36}")),
    ("github_oauth", re.compile(r"gho_[A-Za-z0-9]{36}")),
    ("github_user_to_server", re.compile(r"ghu_[A-Za-z0-9]{36}")),
    ("github_server_to_server", re.compile(r"ghs_[A-Za-z0-9]{36}")),
    ("aws_access_key_id", re.compile(r"\bAKIA[0-9A-Z]{16}\b")),
    # OpenAI-shape last because ``sk-ant-`` would also match a permissive
    # ``sk-...{40,}`` pattern. The earlier anthropic entry will be tried
    # first and reported as ``anthropic_api_key``.
    ("openai_api_key", re.compile(r"\bsk-[A-Za-z0-9]{40,}\b")),
]


def _mask_token(token: str) -> str:
    """Return a masked preview: first 8 chars + ``...`` + last 4."""
    if len(token) <= 12:
        return token  # too short to mask meaningfully
    return f"{token[:8]}...{token[-4:]}"


def _is_binary(content: bytes) -> bool:
    """Heuristic: contains a NUL byte in the first 8KB → binary."""
    return b"\x00" in content[:8192]


def scan_for_secrets(project_dir: Path) -> list[SecretMatch]:
    """Scan staged files for known secret patterns.

    Reads each staged file's CONTENT (``git show :<path>``) rather than
    the working-tree version, so a secret that the user has since
    removed locally but already staged is still caught.

    Returns a list of :class:`SecretMatch` (empty when clean). Binary
    files are skipped — random bytes can match a regex by chance, and
    we don't want to waste time on PNGs.
    """
    staged = _git(project_dir, "diff", "--cached", "--name-only", check=False)
    if staged.returncode != 0:
        return []
    matches: list[SecretMatch] = []
    for path in staged.stdout.splitlines():
        path = path.strip()
        if not path:
            continue
        blob = _git(project_dir, "show", f":{path}", check=False)
        if blob.returncode != 0:
            continue
        raw_bytes = blob.stdout.encode("utf-8", errors="replace")
        if _is_binary(raw_bytes):
            continue
        for lineno, line in enumerate(blob.stdout.splitlines(), start=1):
            for pattern_name, pattern in _SECRET_PATTERNS:
                for m in pattern.finditer(line):
                    matches.append(
                        SecretMatch(
                            file=path,
                            line=lineno,
                            pattern_name=pattern_name,
                            preview=_mask_token(m.group(0)),
                        )
                    )
                    break  # one match per line is enough; the user gets the file:line
    return matches
```

```python
import shutil


@dataclass
class GhCreateResult:
    ok: bool
    repo_url: str | None = None
    error: str | None = None
    install_hint: str | None = None


_GH_INSTALL_HINT = (
    "gh CLI is not installed or not authenticated. Install:\n"
    "  brew install gh   (macOS)\n"
    "  apt install gh    (Debian/Ubuntu)\n"
    "  https://cli.github.com (other platforms)\n"
    "Then authenticate: gh auth login\n"
    "Alternative: create the repo manually on GitHub and re-run\n"
    "  `urika github init <project> --remote <URL>`."
)


def _gh_check_auth() -> bool:
    """Returns True if ``gh auth status`` exits 0 (authenticated)."""
    result = subprocess.run(
        ["gh", "auth", "status"], capture_output=True, text=True
    )
    return result.returncode == 0


def gh_create_repo(project_dir: Path, name: str, public: bool) -> GhCreateResult:
    """Create a GitHub repo via ``gh repo create``, set as origin, push.

    Equivalent to ``gh repo create <name> --{public,private} --source
    <project_dir> --push``. Falls back gracefully when ``gh`` is missing
    or unauthed (no exception, just a structured result with an install
    hint).
    """
    if shutil.which("gh") is None:
        return GhCreateResult(
            ok=False, error="gh CLI not found", install_hint=_GH_INSTALL_HINT
        )
    if not _gh_check_auth():
        return GhCreateResult(
            ok=False,
            error="gh CLI not authenticated",
            install_hint=_GH_INSTALL_HINT,
        )
    visibility = "--public" if public else "--private"
    cmd = [
        "gh", "repo", "create", name,
        visibility,
        "--source", str(project_dir),
        "--push",
    ]
    result = subprocess.run(cmd, capture_output=True, text=True)
    if result.returncode != 0:
        return GhCreateResult(
            ok=False, error=(result.stderr or result.stdout).strip()
        )
    # ``gh repo create`` prints the URL to stdout on success.
    return GhCreateResult(ok=True, repo_url=result.stdout.strip() or None)
```

```python
@click.group("github")

    shells out to the system ``git`` (and optionally ``gh``) CLI."""


@github.command("init")
@click.argument("project", required=False)
@click.option("--remote", default=None, help="Remote URL (e.g. https://github.com/u/r.git)")
@click.option("--create", is_flag=True, help="Use `gh` to create a new GitHub repo")
@click.option("--public", is_flag=True, help="With --create: make the new repo public (default: private)")
def github_init(project, remote, create, public):
    """Initialise the project as a git repo + (optionally) set the remote."""
    from urika.core import git
    project_dir = _resolve_project_dir(project)
    if not project_dir.exists():
        click.secho(f"Unknown project: {project}", fg="red", err=True)
        raise SystemExit(2)

    already = git.is_git_repo(project_dir)
    git.init_repo(project_dir)
    git.write_gitignore(project_dir)

    if remote:
        git.add_remote(project_dir, remote)
        click.echo(f"  remote 'origin' → {remote}")

    if create:
        # T2.2 implements this branch — for now leave as a placeholder
        # that will be filled when T2.2 lands.
        raise NotImplementedError("--create lands in v0.6 T2.2")

    result = git.commit_all(project_dir, message="urika: initial project state")

        click.echo(f"  initial commit {result.sha}")
    elif already:
        click.echo("  already initialised — nothing new to commit")
    else:
```

```python
@github.command("push")
@click.argument("project", required=False)
def github_push(project):
    """Stage everything, commit (after secret scan), and push to origin."""
    from urika.core import git
    project_dir = _resolve_project_dir(project)
    if not git.is_git_repo(project_dir):
        click.secho(f"  not a git repo: {project_dir}", fg="red", err=True)
        click.echo("  run `urika github init <project>` first", err=True)
        raise SystemExit(2)

    # 1. Stage everything so the secret scanner sees what we're about to commit.
    git._git(project_dir, "add", "-A")

    # 2. Secret scan.
    matches = git.scan_for_secrets(project_dir)
    if matches:
        for m in matches:
            click.secho(
                f"  ✗ [urika:secret-detected] {m.file}:{m.line} "
                f"({m.pattern_name})  match: {m.preview}",
                fg="red",
                err=True,
            )
        click.echo(
            "\n  Aborted commit. Remove the secret(s) from the file(s) above "
            "or add the file(s) to .gitignore, then re-run.",
            err=True,
        )
        # Un-stage so the user's tree isn't left in a weird state.
        git._git(project_dir, "reset", "HEAD", check=False)
        raise SystemExit(3)

    # 3. Commit.
    from urika.core.git import recent_commits  # for the summary subject
    runs = "?"
    exp_count = "?"
    # Best-effort subject — read project state for the count
    try:
        from urika.core.progress import _count_runs_and_experiments  # if exists
        runs, exp_count = _count_runs_and_experiments(project_dir)
    except Exception:
        pass
    subject = (
        f"urika: project state ({runs} runs across {exp_count} experiments)"
        if runs != "?"
        else "urika: project state"
    )
    body = "\n\nurika-event: manual\nurika-version: 0.6.0\n"
    commit = git.commit_all(project_dir, message=subject + body)
    if commit.ok:
        click.echo(f"  committed {commit.sha}: {subject}")
    else:
        # No changes to commit is OK — we still try to push (catches the case
        # where a previous commit was never pushed).
        click.echo(f"  (no new commit: {commit.error})")

    pushed = git.push(project_dir)
    if pushed.ok:
```

```python
import tomllib


_EVENT_SUBJECT_TEMPLATES = {
    "run": "urika: experiment run completed",
    "finalize": "urika: finalized project",
    "build-tool": "urika: added/updated project tool",
    "manual": "urika: project state",
}


def _maybe_auto_push(project_dir: Path, event: str) -> bool:
    """If [github].auto_push is true AND ``event`` is in auto_push_after,
    run the same scan→commit→push sequence as ``urika github push`` and
    return True. Returns False (and prints a non-fatal warning where
    appropriate) when skipped or aborted; never raises.

    Used by the tail of ``urika run`` / ``urika finalize`` /
    ``urika build-tool``. The orchestrator stays untouched.
    """
    from urika.core import git as _git_mod

    toml_path = project_dir / "urika.toml"
    if not toml_path.exists():
        return False
    try:
        data = tomllib.loads(toml_path.read_text(encoding="utf-8"))
    except tomllib.TOMLDecodeError:
        return False
    gh = data.get("github") or {}
    if not gh.get("auto_push", False):
        return False
    if event not in gh.get("auto_push_after", []):
        return False
    if not _git_mod.is_git_repo(project_dir):
        click.secho(
            f"  ⚠ auto_push: project not initialised as a git repo "
            f"(run `urika github init {project_dir.name}` first)",
            fg="yellow",
        )
        return False

    # Stage everything for the scanner.
    _git_mod._git(project_dir, "add", "-A")
    matches = _git_mod.scan_for_secrets(project_dir)
    if matches:
        click.secho("  ⚠ urika auto_push: aborted — secret(s) detected:", fg="red")
        for m in matches:
            click.secho(
                f"    [urika:secret-detected] {m.file}:{m.line} "
                f"({m.pattern_name})  match: {m.preview}",
                fg="red",
            )
        click.echo(
            "    Either remove the secret(s) or add to .gitignore, then "
            "run `urika github push` manually."
        )
        _git_mod._git(project_dir, "reset", "HEAD", check=False)
        return False

    subject = _EVENT_SUBJECT_TEMPLATES.get(event, f"urika: {event}")
    body = f"\n\nurika-event: {event}\nurika-version: 0.6.0\n"
    commit = _git_mod.commit_all(project_dir, message=subject + body)
    if not commit.ok:
        # Nothing to commit — still try a push in case earlier commits were
        # never pushed.
        pass

    pushed = _git_mod.push(project_dir)
    if pushed.ok:
        click.echo(f"  ✓ auto_push: {pushed.pushed_sha}")
        return True
    click.secho(
        f"  ⚠ urika auto_push: failed ({pushed.error})", fg="yellow"
    )
    click.echo(
        "    Retry with `urika github push <project>` once the underlying "
        "issue is resolved."
    )
    return False
```

```python
@router.post("/api/projects/{name}/git/push")
async def api_git_push(name: str, request: Request):
    registry = ProjectRegistry().list_all()
    summary = load_project_summary(name, registry)
    if summary is None or summary.missing:
        raise HTTPException(status_code=404, detail="Unknown project")
    from urika.core import git
    if not git.is_git_repo(summary.path):
        raise HTTPException(status_code=409, detail="not a git repo")

    git._git(summary.path, "add", "-A")
    matches = git.scan_for_secrets(summary.path)
    if matches:
        # Unstage to leave the tree in a clean state.
        git._git(summary.path, "reset", "HEAD", check=False)
        return JSONResponse(
            {
                "ok": False,
                "error": (
                    f"secret-detected: {matches[0].file}:{matches[0].line} "
                    f"({matches[0].pattern_name})"
                ),
                "matches": [
                    {"file": m.file, "line": m.line, "pattern": m.pattern_name,
                     "preview": m.preview}
                    for m in matches
                ],
            },
            status_code=409,
        )
    subject = "urika: project state"
    body = "\n\nurika-event: manual\nurika-version: 0.6.0\n"
    git.commit_all(summary.path, message=subject + body)
    pushed = git.push(summary.path)
    return JSONResponse({
        "ok": pushed.ok,
        "pushed_sha": pushed.pushed_sha,
        "error": pushed.error,
    })
```

```python
@router.post("/api/projects/{name}/git/init")
async def api_git_init(name: str, request: Request):
    registry = ProjectRegistry().list_all()
    summary = load_project_summary(name, registry)
    if summary is None or summary.missing:
        raise HTTPException(status_code=404, detail="Unknown project")
    form = await request.form()
    remote = form.get("remote")
    create = (form.get("create") or "").lower() in ("true", "1", "yes")
    if create:
        raise HTTPException(
            status_code=400,
            detail=(
                "--create is only supported via the CLI for v0.6: "
                "run `urika github init <project> --create` in your terminal "
                "where the gh CLI is available."
            ),
        )
    if not remote:
        raise HTTPException(status_code=400, detail="remote URL required")
    from urika.core import git
    git.init_repo(summary.path)
    git.write_gitignore(summary.path)
    git.add_remote(summary.path, remote)
    git.commit_all(summary.path, message="urika: initial project state")
    return JSONResponse({"ok": True, "remote": remote})
```
