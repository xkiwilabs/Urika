    path = _sessions_dir(project_dir) / f"{session_id}.json"

### Task 6.1: `select_best_run` canonical helper

**Files:**
- Create: `src/urika/evaluation/ranking.py`
- Test: `tests/test_evaluation/test_ranking.py` (new)

**Spec:** Single source of truth for "which run is best". Read all four existing implementations first (`labbook.py:342-354 _find_best_run`, `progress.py:50-77 get_best_run`, `leaderboard.py:44-103 update_leaderboard`, `readme_generator.py:198-209 _get_best_metric`) and unify their semantics:

```python
"""Canonical best-run selection — single source of truth.

Pre-v0.7.1, four call sites implemented their own "best run" logic
(labbook, progress, leaderboard, readme_generator) and could disagree
in user-visible artifacts. All four now delegate here.
"""
from __future__ import annotations
from typing import Any

# Union of the direction knowledge previously split across labbook's
# _LOWER_IS_BETTER and leaderboard's direction strings.
LOWER_IS_BETTER: frozenset[str] = frozenset({
    "rmse", "mse", "mae", "mape", "p_value", "aic", "bic", "loss",
    "error", "deviance",
})

def metric_direction(metric: str) -> str:
    """Return "lower" or "higher" for a metric name (suffix-aware)."""
    m = metric.lower()
    if m in LOWER_IS_BETTER or any(m.endswith("_" + s) for s in LOWER_IS_BETTER):
        return "lower"
    return "higher"

def select_best_run(
    runs: list[dict[str, Any]],
    primary_metric: str,
    direction: str | None = None,
) -> dict[str, Any] | None:
    """Best run by *primary_metric*. None if no run has the metric."""
    direction = direction or metric_direction(primary_metric)
    scored = [r for r in runs
              if isinstance(r.get("metrics"), dict)
              and isinstance(r["metrics"].get(primary_metric), (int, float))]
    if not scored:
        return None
    key = lambda r: r["metrics"][primary_metric]
    return min(scored, key=key) if direction == "lower" else max(scored, key=key)
```
