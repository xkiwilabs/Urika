# v0.5.0 — Memory Deepening Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the half-wired project-memory loop — every agent that benefits reads memory, every surface (CLI/REPL/TUI/dashboard) can manage it, and an auto-curator (dry-run by default) keeps the budget in check.

**Architecture:** Build incrementally on `src/urika/core/project_memory.py` (Phase 1 shipped in v0.4). Track 1 adds advisor + report injection and the dashboard memory page. Track 2 adds a `memory_curator` agent role and dry-run auto-trigger driven by a session-scoped flag. Track 3 adds archive + diff views (mtime-driven, no git dep) and a per-project `auto_capture` toggle. Track 4 enforces the token budget and pins cross-surface parity with tests.

**Tech Stack:** Python 3, pytest, FastAPI (dashboard), Textual (TUI), Click (CLI), Claude Agent SDK (agent roles), Jinja2 + HTMX + Alpine (dashboard templates).

**Spec:** `dev/plans/2026-05-17-v0.5.0-memory-deepening.md` (decisions resolved 2026-05-31).

**Conventions:**
- Tests mirror `src/urika/` under `tests/`. Shared fixtures in `tests/conftest.py` (`tmp_project_dir`, `tmp_urika_home`).
- Run unit suite with: `URIKA_SKIP_SMOKE=1 pytest -q -m "not integration"`.
- Lint/format: `ruff check src/ tests/ && ruff format src/ tests/`.
- One commit per task. No co-author trailer (public repo — see memory `feedback_no_coauthor_public`).
- No pre-commit hook (only pre-push); commit normally.

**Decisions baked into this plan:**
1. Report agent injection — included in Track 1.
2. Token counting — `bytes / 3.5` conservative byte-heuristic.
3. Diff source — filesystem mtime.
4. Curator auto-trigger — once per session, **dry-run by default**.

```python
def load_project_memory(project_dir: Path) -> str:
    """Return memory contents as a single markdown string.

    Hard-cap behaviour (Decision 4): when the assembled body would
    exceed HARD_CAP_TOKENS, truncate from the OLDEST entries forward
    (keep the newest, which are most likely to be relevant), and
    append a marker the user sees in the agent's text output so they
    know to run the curator.
    """
    if not is_enabled(project_dir):
        return ""

    entries = list_entries(project_dir)  # already ordered (e.g. by mtime)
    if not entries:
        return ""

    # Build full body first to compute size cheaply.
    md = memory_dir(project_dir)
    pieces: list[tuple[str, str]] = []  # (filename, rendered)
    for entry in entries:
        path = md / entry["filename"]
        try:
            text = path.read_text(encoding="utf-8")
        except OSError:
            continue
        pieces.append((entry["filename"], text))

    # Keep newest-first; drop from the end if over hard cap.
    kept_pieces: list[str] = []
    total = 0
    overflow = False
    for _name, text in reversed(pieces):  # newest first
        t = estimate_tokens(text)
        if total + t > HARD_CAP_TOKENS:
            overflow = True
            break
        kept_pieces.append(text)
        total += t
    kept_pieces.reverse()  # restore original (chronological) order

    body = "\n\n".join(kept_pieces)
    if overflow:
        body += (
            "\n\n## Project Memory (over budget — run urika memory curate)\n"
            "The memory directory is past the hard cap "
            f"({HARD_CAP_TOKENS} tokens). The oldest entries were "
            "truncated from this injection. Run `urika memory curate` "
            "to compact the directory.\n"
        )
    return body
```

> The exact "reversed → take while fits → re-reverse" pattern matters: newest entries are most relevant and must be kept; the OLDEST get dropped first. If the current `list_entries` ordering is reverse of expected, adapt — but test against the new behaviour, not the old.

```python
"""Memory curator dispatch — invokes the agent, parses decisions,
optionally applies them."""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path
from typing import Any

from urika.core import project_memory


@dataclass
class CuratorResult:
    decisions: list[dict[str, Any]] = field(default_factory=list)
    applied: bool = False
    error: str | None = None


def _build_curator_input(project_dir: Path) -> str:
    """Render the current memory dir as the curator's user message."""
    entries = project_memory.list_entries(project_dir)
    md = project_memory.memory_dir(project_dir)
    parts = ["# Current memory directory\n"]
    for entry in entries:
        path = md / entry["filename"]
        try:
            text = path.read_text(encoding="utf-8")
        except OSError:
            continue
        parts.append(f"---\n## {entry['filename']}\n\n{text}\n")
    return "\n".join(parts)


async def run_curator(project_dir: Path, runner, apply: bool = False) -> CuratorResult:
    """Run the memory curator. Returns proposed decisions.

    apply=False (default): dry-run — return decisions, mutate nothing.
    apply=True: also commits the decisions to the memory dir (Task 2.6).
    """
    from urika.agents.roles import memory_curator as role
    from urika.orchestrator.parsing import _extract_json_blocks

    user_message = _build_curator_input(project_dir)
    raw = await runner.run(role.build_config(), user_message)

    blocks = _extract_json_blocks(raw)
    decisions: list[dict[str, Any]] = []
    for block in blocks:
        d = block.get("decisions")
        if isinstance(d, list):
            decisions.extend(d)

    result = CuratorResult(decisions=decisions, applied=False)
    if apply:
        # Filled in by Task 2.6
        pass
    return result
```

```python
def archive_entry(project_dir: Path, filename: str) -> Path:
    """Move an entry to memory/archive/<YYYY-MM>/<filename>. Returns the

    archive_root = memory_dir(project_dir) / "archive" / yyyy_mm
    archive_root.mkdir(parents=True, exist_ok=True)

    src.replace(dst)
    rebuild_index(project_dir)
    return dst
```

```python
def _apply_decisions(project_dir: Path, decisions: list[dict[str, Any]]) -> bool:
    from urika.core.project_memory import (
        archive_entry,
        memory_dir,
        rebuild_index,
        _split_frontmatter,
    )

    md = memory_dir(project_dir)
    for d in decisions:
        action = d.get("action")
        try:
            if action == "archive":
                fn = d.get("filename")
                if isinstance(fn, str):
                    archive_entry(project_dir, fn)
            elif action == "merge":
                keep = d.get("into")
                if not isinstance(keep, str):
                    continue
                from_list = [f for f in d.get("from", []) if isinstance(f, str)]
                _apply_merge(md, keep, from_list)
                for fn in from_list:
                    archive_entry(project_dir, fn)
            elif action in ("flag_stale", "flag_contradiction"):
                _apply_flag(md, d)
        except Exception:  # noqa: BLE001 — decision-level errors don't abort the batch
            continue
    rebuild_index(project_dir)
    return True


def _apply_merge(memory_dir_path: Path, keep: str, from_list: list[str]) -> None:
    keep_path = memory_dir_path / keep
    if not keep_path.is_file():
        return
    keep_text = keep_path.read_text(encoding="utf-8")
    add: list[str] = []
    for fn in from_list:
        src = memory_dir_path / fn
        if not src.is_file():
            continue
        add.append(src.read_text(encoding="utf-8"))
    if add:
        keep_path.write_text(keep_text + "\n\n---\n\n" + "\n\n---\n\n".join(add), encoding="utf-8")


def _apply_flag(memory_dir_path: Path, decision: dict) -> None:
    fn = decision.get("filename") or (decision.get("between") or [None])[0]
    if not isinstance(fn, str):
        return
    path = memory_dir_path / fn
    if not path.is_file():
        return
    flag_key = "flagged_stale" if decision["action"] == "flag_stale" else "flagged_contradiction"
    reason = decision.get("reason", "")
    text = path.read_text(encoding="utf-8")
    # Idempotent: don't add the same flag twice.
    if f"{flag_key}:" in text:
        return
    # Inject into frontmatter (between leading --- ... ---) or prepend a stub.
    lines = text.splitlines()
    if lines and lines[0] == "---":
        # find closing ---
        for i, line in enumerate(lines[1:], start=1):
            if line == "---":
                lines.insert(i, f"{flag_key}: {reason}")
                break
    else:
        lines = ["---", f"{flag_key}: {reason}", "---"] + lines
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
```

```python
async def _maybe_run_curator(
    project_dir: Path, runner, session_id: str | None
) -> str | None:
    """If the session-scoped curator flag is set, run the curator in
    dry-run, mark this session done, and return a notice the orchestrator
    surfaces to the agent's text output. Returns None when nothing to do."""
    from urika.core import project_memory
    from urika.core.memory_curator import run_curator

    if not session_id or not project_memory.is_curator_due(project_dir, session_id):
        return None
    try:
        result = await run_curator(project_dir, runner=runner, apply=False)
    except Exception as exc:  # noqa: BLE001 — curator failure must not break the turn
        project_memory.mark_curator_ran(project_dir, session_id)
        return f"Memory curator: failed to run ({exc}). Skipping."

    project_memory.mark_curator_ran(project_dir, session_id)
    if not result.decisions:
        return "Memory curator: ran (no changes suggested)."

    lines = [f"Memory curator: {len(result.decisions)} suggestion(s). Run `urika memory curate --apply` to commit."]
    for d in result.decisions[:5]:
        lines.append(f"  - {d.get('action')}: {d.get('reason', '')}")
    if len(result.decisions) > 5:
        lines.append(f"  ... and {len(result.decisions) - 5} more.")
    return "\n".join(lines)
```

```python
def list_archive(project_dir: Path, month: str | None = None) -> list[dict[str, str]]:
    """List archived entries, optionally filtered to one YYYY-MM month."""
    root = memory_dir(project_dir) / "archive"
    if not root.is_dir():
        return []
    out: list[dict[str, str]] = []
    months = [month] if month else sorted(p.name for p in root.iterdir() if p.is_dir())
    for m in months:
        mdir = root / m
        if not mdir.is_dir():
            continue
        for p in sorted(mdir.iterdir()):
            if p.suffix == ".md":
                fm, _ = _split_frontmatter(p.read_text(encoding="utf-8"))
                out.append({
                    "filename": p.name,
                    "month": m,
                    "title": fm.get("title", p.stem),
                    "type": fm.get("type", _infer_type_from_filename(p.name)),
                })
    return out


def diff_between(
    project_dir: Path,
    from_ts: datetime,
    to_ts: datetime,
) -> dict[str, list[dict[str, str]]]:
    """Classify entries by mtime between two timestamps.

    Returns {'added': [...], 'archived': [...], 'trashed': [...]}.
    'changed' is not detectable from mtime alone — a single mtime
    means "last touched" with no notion of prior state — so we omit it.
    Per-edit history is a v0.6.0+ concern (GitHub auto-backup).
    """
    md = memory_dir(project_dir)
    added: list[dict[str, str]] = []
    archived: list[dict[str, str]] = []
    trashed: list[dict[str, str]] = []

    def _in_window(p: Path) -> bool:
        st = p.stat().st_mtime
        return from_ts.timestamp() <= st <= to_ts.timestamp()

    if md.is_dir():
        for p in md.iterdir():
            if p.is_file() and p.suffix == ".md" and _in_window(p):
                added.append({"filename": p.name})
    archive_root = md / "archive"
    if archive_root.is_dir():
        for sub in archive_root.iterdir():
            if not sub.is_dir():
                continue
            for p in sub.iterdir():
                if p.is_file() and p.suffix == ".md" and _in_window(p):
                    archived.append({"filename": p.name, "month": sub.name})
    trash_root = md / ".trash"
    if trash_root.is_dir():
        for p in trash_root.iterdir():
            if p.is_file() and p.suffix == ".md" and _in_window(p):
                trashed.append({"filename": p.name})
    return {"added": added, "archived": archived, "trashed": trashed}
```

```python
def _parse_relative(s: str) -> datetime:
    """Parse 'now', 'Nh ago', 'Nd ago', 'Nw ago', or ISO 8601."""
    from datetime import datetime, timedelta, timezone
    s = s.strip().lower()
    now = datetime.now(timezone.utc)
    if s == "now":
        return now
    if s == "yesterday":
        return now - timedelta(days=1)
    import re
    m = re.match(r"^(\d+)([hdw])\s+ago$", s)
    if m:
        n = int(m.group(1))
        unit = m.group(2)
        return now - {"h": timedelta(hours=n), "d": timedelta(days=n), "w": timedelta(weeks=n)}[unit]
    # ISO 8601
    return datetime.fromisoformat(s)
```
