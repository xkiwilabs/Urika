The Claude adapter uses the SDK's built-in tools (Bash/Read/Write/etc)

    output: str
    error: str | None = None

    @staticmethod
    def for_role(role: str) -> list[str]:

        return ["bash", "read", "write", "grep", "glob", "edit"]

# src/urika/agents/tools/security.py
"""SecurityPolicy enforcement helpers shared across tools.

Mirrors the runtime checks in `urika.agents.permission` so adapters
that don't go through the Claude SDK's can_use_tool callback still
get the same enforcement.
"""

from __future__ import annotations

import shlex
from pathlib import Path

from urika.agents.permission import SecurityPolicy


def enforce_bash(cmd: str, policy: SecurityPolicy) -> str | None:
    """Return None if `cmd` is allowed, else a human-readable reason."""
    try:
        tokens = shlex.split(cmd)
    except ValueError as exc:
        return f"unparseable command: {exc}"
    if not tokens:
        return "empty command"
    head = tokens[0]
    if head not in policy.allowed_bash_commands:
        return f"command {head!r} not in allowlist"
    for tok in tokens:
        if any(ch in tok for ch in ";|&`$()<>"):
            return f"shell metacharacter rejected in token {tok!r}"
    return None


def enforce_path_write(path: str, policy: SecurityPolicy) -> tuple[Path | None, str | None]:
    """Resolve `path` and check it's under a writable dir. Returns (resolved, error)."""
    p = Path(path).expanduser().resolve(strict=False)
    for writable in policy.writable_dirs:
        try:
            p.relative_to(Path(writable).expanduser().resolve(strict=False))
            return p, None
        except ValueError:
            continue
    return None, f"path {p} not under any writable dir"


def enforce_path_read(path: str, policy: SecurityPolicy) -> tuple[Path | None, str | None]:
    """Resolve `path` and check it's under a readable dir. Returns (resolved, error)."""
    p = Path(path).expanduser().resolve(strict=False)
    readable = list(policy.readable_dirs) + list(policy.writable_dirs)
    for d in readable:
        try:
            p.relative_to(Path(d).expanduser().resolve(strict=False))
            return p, None
        except ValueError:
            continue
    return None, f"path {p} not under any readable dir"
```

# src/urika/agents/tools/bash.py
"""Bash tool callable by adapters that need a function-shaped wrapper."""

from __future__ import annotations

import subprocess
from pathlib import Path

from urika.agents.permission import SecurityPolicy
from urika.agents.tools import ToolResult
from urika.agents.tools.security import enforce_bash


def bash(
    cmd: str,
    *,
    security: SecurityPolicy,
    project_dir: Path,
    timeout_seconds: int = 600,
) -> ToolResult:
    deny = enforce_bash(cmd, security)
    if deny is not None:
        return ToolResult(ok=False, output="", error=deny, exit_code=None)
    try:
        proc = subprocess.run(
            cmd,
            shell=True,
            cwd=str(project_dir),
            capture_output=True,
            text=True,
            timeout=timeout_seconds,
            check=False,
        )
    except subprocess.TimeoutExpired:
        return ToolResult(
            ok=False, output="", error=f"timed out after {timeout_seconds}s", exit_code=None
        )
    return ToolResult(
        ok=proc.returncode == 0,
        output=proc.stdout,
        error=proc.stderr if proc.returncode != 0 else None,
        exit_code=proc.returncode,
    )
```

from urika.agents.permission import SecurityPolicy
from urika.agents.tools import ToolResult
from urika.agents.tools.security import enforce_path_read


def read(path: str, *, security: SecurityPolicy) -> ToolResult:
    resolved, err = enforce_path_read(path, security)
    if err:
        return ToolResult(ok=False, output="", error=err)
    try:
        text = resolved.read_text(encoding="utf-8")
    except FileNotFoundError:
        return ToolResult(ok=False, output="", error=f"no such file: {resolved}")
    except OSError as exc:
        return ToolResult(ok=False, output="", error=str(exc))
    return ToolResult(ok=True, output=text)

# src/urika/agents/tools/grep.py
from __future__ import annotations

import re
from pathlib import Path

from urika.agents.permission import SecurityPolicy
from urika.agents.tools import ToolResult
from urika.agents.tools.security import enforce_path_read


def grep(pattern: str, path: str, *, security: SecurityPolicy) -> ToolResult:
    resolved, err = enforce_path_read(path, security)
    if err:
        return ToolResult(ok=False, output="", error=err)
    try:
        rx = re.compile(pattern)
    except re.error as exc:
        return ToolResult(ok=False, output="", error=f"bad pattern: {exc}")
    hits: list[str] = []
    for fp in _walk_files(resolved):
        try:
            for n, line in enumerate(fp.read_text(encoding="utf-8", errors="ignore").splitlines(), 1):
                if rx.search(line):
                    hits.append(f"{fp}:{n}:{line}")
        except OSError:
            continue
    return ToolResult(ok=True, output="\n".join(hits))


def _walk_files(root: Path):
    if root.is_file():
        yield root
        return
    for p in root.rglob("*"):
        if p.is_file():
            yield p
```

from __future__ import annotations

from urika.agents.permission import SecurityPolicy
from urika.agents.tools import ToolResult
from urika.agents.tools.security import enforce_path_write


def edit(
    path: str,
    old_string: str,
    new_string: str,
    *,
    security: SecurityPolicy,
) -> ToolResult:
    resolved, err = enforce_path_write(path, security)
    if err:
        return ToolResult(ok=False, output="", error=err)
    try:
        text = resolved.read_text(encoding="utf-8")
    except OSError as exc:
        return ToolResult(ok=False, output="", error=str(exc))
    count = text.count(old_string)
    if count == 0:
        return ToolResult(ok=False, output="", error="old_string not found")
    if count > 1:
        return ToolResult(
            ok=False, output="", error=f"old_string occurs {count} times — must be unique"
        )
    resolved.write_text(text.replace(old_string, new_string), encoding="utf-8")
    return ToolResult(ok=True, output=f"edited {resolved}")

# src/urika/agents/pricing.py
"""Single source of $-per-1M-tokens truth across all adapters.

Gemini prices are placeholders — verify against published rates
before tagging v0.7.0.
"""

from __future__ import annotations

PRICES: dict[str, tuple[float, float]] = {
    # Anthropic — (input $/1M, output $/1M)
    "claude-opus-4-7": (15.0, 75.0),
    "claude-sonnet-4-5": (3.0, 15.0),
    # OpenAI
    "gpt-5.4": (2.50, 15.0),
    "gpt-5.4-mini": (0.75, 4.50),
    "gpt-5.4-nano": (0.20, 1.25),
    "gpt-5.4-pro": (30.0, 180.0),
    "gpt-5.4-thinking": (2.50, 15.0),
    # Google — placeholders, verify at impl time
    "gemini-3-pro": (5.0, 25.0),
    "gemini-3-flash": (0.50, 2.50),
}


def cost_for(model: str, input_tokens: int, output_tokens: int) -> float:
    rates = PRICES.get(model)
    if rates is None:
        return 0.0
    in_rate, out_rate = rates
    return (input_tokens / 1_000_000) * in_rate + (output_tokens / 1_000_000) * out_rate
```

# src/urika/agents/registry.py
"""Model-prefix → AgentRunner routing.

The orchestrator calls `runner_for(model_string)` to get the
adapter that should handle a particular model. Routing is by name
prefix — there's no per-project provider setting, the model string
itself is the source of truth.
"""

from __future__ import annotations

from urika.agents.runner import AgentRunner


class UnknownModelError(ValueError):
    """Raised when no adapter is registered for a model prefix."""


def runner_for(model: str) -> AgentRunner:
    if model.startswith("claude-"):
        from urika.agents.adapters.claude_sdk import ClaudeSDKRunner

        return ClaudeSDKRunner()
    if model.startswith(("gpt-", "o3", "o4", "o5")):
        from urika.agents.adapters.openai_agents import OpenAIAgentsRunner

        return OpenAIAgentsRunner()
    if model.startswith("gemini-"):
        from urika.agents.adapters.google_adk import GoogleADKRunner

        return GoogleADKRunner()
    raise UnknownModelError(f"no adapter for model {model!r}")
```

    async def _run_with_sdk(self, config: AgentConfig, prompt: str) -> AgentResult:
        import agents as openai_agents_sdk
        from urika.agents.pricing import cost_for
        from urika.agents.tools import ToolRegistry
        from urika.agents.tools.bash import bash as bash_tool
        from urika.agents.tools.read import read as read_tool
        from urika.agents.tools.write import write as write_tool
        from urika.agents.tools.grep import grep as grep_tool
        from urika.agents.tools.glob import glob as glob_tool
        from urika.agents.tools.edit import edit as edit_tool

        security = config.security_policy
        project_dir = getattr(config, "project_dir", None)

        @openai_agents_sdk.function_tool
        def bash(cmd: str) -> str:
            r = bash_tool(cmd, security=security, project_dir=project_dir)
            return r.output if r.ok else f"ERROR: {r.error}"

        @openai_agents_sdk.function_tool
        def read(path: str) -> str:
            r = read_tool(path, security=security)
            return r.output if r.ok else f"ERROR: {r.error}"

        @openai_agents_sdk.function_tool
        def write(path: str, content: str) -> str:
            r = write_tool(path, content, security=security)
            return r.output if r.ok else f"ERROR: {r.error}"

        @openai_agents_sdk.function_tool
        def grep(pattern: str, path: str) -> str:
            r = grep_tool(pattern, path, security=security)
            return r.output if r.ok else f"ERROR: {r.error}"

        @openai_agents_sdk.function_tool
        def glob(pattern: str) -> str:
            r = glob_tool(pattern, security=security, project_dir=project_dir)
            return r.output if r.ok else f"ERROR: {r.error}"

        @openai_agents_sdk.function_tool
        def edit(path: str, old_string: str, new_string: str) -> str:
            r = edit_tool(path, old_string, new_string, security=security)
            return r.output if r.ok else f"ERROR: {r.error}"

        tools_by_name = {
            "bash": bash, "read": read, "write": write,
            "grep": grep, "glob": glob, "edit": edit,
        }
        tools = [tools_by_name[n] for n in ToolRegistry.for_role(config.name)]

        agent = openai_agents_sdk.Agent(
            name=config.name,
            instructions=config.system_prompt,
            model=config.model,
            tools=tools,
        )
        max_turns = getattr(config, "max_turns", 10)

        try:
            result = await openai_agents_sdk.Runner.run(agent, prompt, max_turns=max_turns)
        except Exception as exc:  # noqa: BLE001 — SDK boundary
            return AgentResult(
                text_output="", success=False, usage=None, error=str(exc)
            )

        usage_obj = getattr(result, "usage", None)
        in_tok = getattr(usage_obj, "input_tokens", 0) or 0
        out_tok = getattr(usage_obj, "output_tokens", 0) or 0
        usage = {
            "input_tokens": in_tok,
            "output_tokens": out_tok,
            "cost_usd": cost_for(config.model, in_tok, out_tok),
            "mode": "api_key",
        }
        return AgentResult(
            text_output=getattr(result, "final_output", "") or "",
            success=True,
            usage=usage,
            error=None,
        )

# src/urika/cli/config_provider.py
"""`urika config anthropic|openai|google` — provider auth setup."""

from __future__ import annotations

import click

from urika.core.settings import set_preference


@click.command("anthropic")
@click.option("--mode", type=click.Choice(["api_key", "subscription"]), required=True)
@click.option("--no-verify", is_flag=True, default=False)
def config_anthropic(mode: str, no_verify: bool) -> None:
    """Configure Anthropic authentication (api_key or subscription)."""
    set_preference("anthropic_auth", mode)
    if mode == "subscription":
        click.echo(
            "Anthropic subscription mode enabled. Run `claude login` in your terminal "
            "to authenticate the local Claude CLI before running experiments."
        )
        if not no_verify:
            click.echo("Verifying `claude --version`...")
            import subprocess

            try:
                proc = subprocess.run(
                    ["claude", "--version"], capture_output=True, text=True, check=False
                )
                if proc.returncode != 0:
                    click.echo(
                        f"⚠ claude CLI not available: {proc.stderr.strip()}",
                        err=True,
                    )
            except FileNotFoundError:
                click.echo(
                    "⚠ claude CLI not found on PATH — install per Anthropic's docs.",
                    err=True,
                )
    else:
        click.echo("Anthropic API-key mode set. Ensure ANTHROPIC_API_KEY is configured.")


@click.command("openai")
@click.option("--mode", type=click.Choice(["api_key", "subscription"]), required=True)
def config_openai(mode: str) -> None:
    if mode == "subscription":
        raise click.UsageError(
            "OpenAI Plus subscriptions cannot be used with Urika. OpenAI's API key is "
            "a separate product — get one at https://platform.openai.com/api-keys."
        )
    set_preference("openai_auth", "api_key")
    click.echo("OpenAI API-key mode set. Run `urika config api-key OPENAI_API_KEY` to store it.")


@click.command("google")
@click.option("--mode", type=click.Choice(["api_key", "subscription"]), required=True)
def config_google(mode: str) -> None:
    if mode == "subscription":
        raise click.UsageError(
            "Google AI Pro/Ultra subscriptions cannot be used with Urika — Google's "
            "Terms of Service forbid third-party use. Get a Gemini API key from "
            "https://aistudio.google.com/app/apikey."
        )
    set_preference("google_auth", "api_key")
    click.echo("Google API-key mode set. Run `urika config api-key GEMINI_API_KEY` to store it.")
```
